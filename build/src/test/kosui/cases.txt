# Cases for the kosui suite.
#
#   echo <input>   input line in echo mode (evaluated as a sentence)
#   cmd <input>    input line in command mode (evaluated as $(<input>))
#   = <text>       expected output, one line per "=" line (a lone "=" is an empty line)
#
# The output must match exactly. Every case runs in a fresh copy of the
# fixture directory, so a case never sees the changes of another one.
# Use ASCII only, and no entry whose value has more than one word (random).

# entry reference
echo ${greet}
= Hello

echo ${nest}
= Hello world!

echo ${${key}}
= world

echo plain text
= plain text

# an undefined entry gives an empty string
echo ${nosuch}
=

# expression
echo $[ 1 + 2 ]
= 3

echo $[ ${x} + 1 ]
= 2

# inline script
echo $(echo hi)
= hi

# command mode
cmd echo hi
= hi

cmd size npw
= 3

# gsub: the pattern is not found after the last match (npos handling)
echo $(gsub "a-b-c" "-" "+")
= a+b+c

echo $(gsub "abc" "x" "y")
= abc

echo $(gsub "aXbXc" "X" "")
= abc

# tr: a char is replaced, a char without a counterpart is deleted
echo $(tr "hello" "l" "L")
= heLLo

echo $(tr "abc" "x" "y")
= abc

# substr: a huge length must not overflow
echo $(substr "hello" 1 2147483647)
= ello

echo $(substr "hello" 1 2)
= el

# split: the last piece has no delimiter after it
echo $(split sp "a-b-c" "-")$(size sp)
= 3

echo $(split sp2 "abc" "-")$(size sp2)
= 1

# pop on an entry that was emptied by clear
echo $(push t a)$(clear t)$(pop t)ok$(size t)
= ok0

echo $(push u a)$(push u b)$(pop u)$(size u)
= b1

# ---- regex_* (uka_regexp) ----
echo $(regex_match "abc123def" "[0-9]+" "" m)$(join m "|")
= 1123

echo $(regex_match "abc" "[0-9]+" "" m)$(size m)
= 00

echo $(regex_match "key=42" "([a-z]+)=([0-9]+)" "" m)$(join m "|")
= 1key=42|key|42

# the entry is replaced, not appended to
echo $(regex_match "a1" "[0-9]" "" m)$(regex_match "b2" "[0-9]" "" m)$(size m)
= 111

echo $(regex_match "ABC" "abc" "i")$(regex_match "ABC" "abc")
= 10

echo $(regex_match "ab" "a b" "x")$(regex_match "ab" "a b")$(regex_match "ab" "A B" "ix")
= 101

echo $(regex_find "abc123def" "[0-9]")
= 3

echo $(regex_find "a1b2" "[0-9]" "" 2)$(regex_find "a1b2" "[0-9]" "" -1)$(regex_find "a1b2" "[0-9]" "" 9)
= 33-1

echo $(regex_find "ab12" "([a-z]+)([0-9]+)" "" "" f)$(join f "|")
= 0ab12|ab|12

echo $(regex_find "abc" "[0-9]")
= -1

echo $(regex_findall fa "a1b22c333" "[0-9]+")$(join fa "|")
= 31|22|333

echo $(regex_count "a1b22c333" "[0-9]+")$(regex_count "abc" "[0-9]")
= 30

echo $(regex_replace "a1b22" "([a-z])([0-9]+)" "$2$1")
= 1a22b

echo $(regex_replace_first "a1b22" "[0-9]" "#")$(regex_replace "a1b22" "[0-9]" "#")
= a#b22a#b##

echo $(regex_erase "a1b22" "[0-9]")$(regex_erase_first "a1b22" "[0-9]")
= abab22

echo $(regex_replace "abc" "x" "y")
= abc

echo $(regex_split sp "a, b,c" ",\s*")$(join sp "|")
= 3a|b|c

echo $(regex_split sp "a,b,c,d" "," "" 2)$(join sp "|")
= 2a|b,c,d

echo $(regex_split sp "abc" "")$(join sp "|")
= 3a|b|c

echo $(regex_escape "a.b*c(d)")
= a\.b\*c\(d\)

# subroutine calls: a name the pattern does not define is the expansion of ${name}
echo $(regex_match "2026-10" "^(?&re_date)$")$(regex_match "2026-1" "^(?&re_date)$")
= 10

# the groups of the subroutines are not in the result, only those of the pattern
echo $(regex_match "x2026-10y" "(x)(?&re_date)(y)" "" g)$(join g "|")
= 1x2026-10y|x|y

echo $(regex_match "key=42" "(?&re_pair)" "" g)$(size g)
= 11

echo $(regex_match "key=42" "^(\w+)=(?&re_digits)$" "" g)$(join g "|")
= 1key=42|key

echo $(regex_replace "a2026-10b" "(?&re_date)" "<$&>")
= a<2026-10>b

# errors give an empty string
echo [$(regex_match "abc" "[")][$(regex_match "abc" "(?&nosuch)")][$(regex_match "abc" "a" "z")]
= [][][]

# a subroutine that calls itself on the left does not hang
echo $(regex_match "xxx" "(?&re_loop)")
= 0

# lambda: an anonymous function named System.Lambda.<n>
echo $(set f $(lambda $(echo [$@arg[1]])))${f}
= System.Lambda.1

echo $(set f $(lambda $(echo [$@arg[1]])))$(${f} hi)
= [hi]

# each lambda gets its own name
echo $(set f $(lambda $(echo a)))$(set g $(lambda $(echo b)))$(${f})$(${g})
= ab

# a lambda is removed with "rmfunc" like any user function
echo $(set f $(lambda $(echo a)))$(rmfunc ${f})[$(${f})]
= []

# (*UCP) is accepted by the engine; an unknown verb is still a syntax error
echo [$(regex_match "abc" "(*UCP)^\w+$")][$(regex_match "abc" "(*NOSUCH)a")]
= [1][]

# textsave_charset / textload_charset: an ASCII text survives a round trip in every charset
# (the multibyte round trips are not here because the cases are ASCII only)
echo $(textsave_charset t.txt Shift_JIS greet)$(textload_charset r t.txt SJIS)[${r}]
= [Hello]

echo $(textsave_charset t.txt EUC-JP greet)$(textload_charset r t.txt euc-jp)[${r}]
= [Hello]

echo $(textsave_charset t.txt ISO-2022-JP greet)$(textload_charset r t.txt ISO-2022-JP)[${r}]
= [Hello]

echo $(textsave_charset t.txt EUC-KR greet)$(textload_charset r t.txt EUC-KR)[${r}]
= [Hello]

echo $(textsave_charset t.txt GB2312 greet)$(textload_charset r t.txt GB2312)[${r}]
= [Hello]

# several entries are written one line per word; "auto" detects the charset like textload
echo $(textsave_charset t.txt UTF-8 greet name)$(textload_charset r t.txt auto)$(size r)$(join r "|")
= 2Hello|world

# textappend_charset keeps the old contents
echo $(textsave_charset t.txt UTF-8 greet)$(textappend_charset t.txt GB2312 name)$(textload_charset r t.txt auto)$(size r)$(join r "|")
= 2Hello|world

# an unknown charset does nothing: the file is not overwritten and the entry is not changed
echo $(textsave_charset t.txt UTF-8 greet)$(textsave_charset t.txt NOSUCH name)$(textappend_charset t.txt NOSUCH name)$(textload_charset r t.txt UTF-8)$(textload_charset q t.txt NOSUCH)$(size r)[${r}][${q}]
= 1[Hello][]

# textload / textsave keep their old forms
echo $(textsave t.txt greet)$(textload r t.txt)[${r}]
= [Hello]
